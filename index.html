<script>

const analytics = getAnalytics(app);


// Initialiser Firebase

const app = firebase.initializeApp(firebaseConfig);

const db = firebase.firestore(app);


// ===== LES VARIABLES =====

let nombreSecret;

let tentatives;

let jeuTermine;

let pseudoJoueur = localStorage.getItem("pseudoJoueur") || "Anonyme";


// ===== AFFICHER LE PSEUDO =====

function afficherPseudo() {

document.getElementById("pseudoAffiche").textContent = `Pseudo : ${pseudoJoueur}`;

}


// ===== CHANGER LE PSEUDO =====

function changerPseudo() {

const nouveau = document.getElementById("pseudoInput").value.trim();

if (nouveau && nouveau.length > 0) {

pseudoJoueur = nouveau;

localStorage.setItem("pseudoJoueur", pseudoJoueur);

afficherPseudo();

document.getElementById("pseudoInput").value = "";

chargerScores();

} else {

alert("Entre un pseudo valide !");

}

}


// ===== FONCTION DE DÉMARRAGE =====

function demarrer() {

nombreSecret = Math.floor(Math.random() * 100) + 1;

tentatives = 0;

jeuTermine = false;

document.getElementById("leMessage").textContent = "";

document.getElementById("monEntree").value = "";

document.getElementById("etat").textContent = "En jeu...";

document.getElementById("compteur").textContent = "0";

document.getElementById("monEntree").focus();

}


// ===== VÉRIFIER LA RÉPONSE =====

function verifierReponse() {

if (jeuTermine) return;


let entree = document.getElementById("monEntree");

let nombre = parseInt(entree.value);


if (isNaN(nombre) || nombre < 1 || nombre > 100) {

afficherMessage("⚠️ Écris un nombre entre 1 et 100 !", "indice");

return;

}


tentatives++;

document.getElementById("compteur").textContent = tentatives;


if (nombre === nombreSecret) {

afficherMessage(

`🎉 BRAVO ! Tu as trouvé ${nombreSecret} en ${tentatives} tentative(s) !`,

"victoire"

);

document.getElementById("etat").textContent = "Gagné ! 🎉";

jeuTermine = true;

// Sauvegarder le score dans Firebase

sauvegarderScore(tentatives);

} else if (nombre < nombreSecret) {

afficherMessage(`📈 Le nombre est PLUS GRAND que ${nombre}`, "indice");

} else {

afficherMessage(`📉 Le nombre est PLUS PETIT que ${nombre}`, "indice");

}


entree.value = "";

entree.focus();

}


// ===== SAUVEGARDER LE SCORE DANS FIREBASE =====

async function sauvegarderScore(score) {

try {

afficherMessage("💾 Sauvegarde en cours...", "loading");

await db.collection("scores").add({

joueur: pseudoJoueur,

score: score,

nombreSecret: nombreSecret,

date: new Date(),

timestamp: firebase.firestore.FieldValue.serverTimestamp()

});


afficherMessage("✅ Score sauvegardé !", "victoire");

// Recharger les scores

setTimeout(chargerScores, 500);

} catch (error) {

console.error("Erreur sauvegarde :", error);

afficherMessage("❌ Erreur sauvegarde", "indice");

}

}


// ===== CHARGER LES SCORES DEPUIS FIREBASE =====

async function chargerScores() {

try {

const snapshot = await db.collection("scores")

.orderBy("score")

.limit(10)

.get();


const container = document.getElementById("scoresContainer");


if (snapshot.empty) {

container.innerHTML = '<div class="no-scores">Aucun score pour l\'instant. À toi de jouer ! 🎮</div>';

return;

}


let html = "";

snapshot.forEach((doc, index) => {

const data = doc.data();

html += `

<div class="score-item">

<span class="score-joueur">${index + 1}. ${data.joueur}</span>

<span class="score-value">${data.score} tentatives</span>

</div>

`;

});


container.innerHTML = html;

} catch (error) {

console.error("Erreur chargement :", error);

document.getElementById("scoresContainer").innerHTML =

'<div class="no-scores">Erreur de connexion à Firebase</div>';

}

}


// ===== AFFICHER LES MESSAGES =====

function afficherMessage(texte, style) {

let messageElement = document.getElementById("leMessage");

messageElement.textContent = texte;

messageElement.className = "message " + style;

}


// ===== RECOMMENCER =====

function recommencer() {

demarrer();

chargerScores();

}


// ===== APPUYER SUR ENTRÉE =====

document.getElementById("monEntree").addEventListener("keypress", function(event) {

if (event.key === "Enter") {

verifierReponse();

}

});


document.getElementById("pseudoInput").addEventListener("keypress", function(event) {

if (event.key === "Enter") {

changerPseudo();

}

});


// ===== AU DÉMARRAGE =====

afficherPseudo();
